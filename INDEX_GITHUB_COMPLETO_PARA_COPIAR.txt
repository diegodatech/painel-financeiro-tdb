    #splash{position:fixed;inset:0;z-index:20;background:var(--paper);display:flex;align-items:center;justify-content:center;padding:24px;text-align:center;transition:opacity .12s ease,visibility .12s ease}

  <div id="splash" aria-live="polite">
    <div class="conteudo">
      <img class="logo" src="./logo.png?v=13" alt="Escola Infantil Tempo de Brincar">
      <div class="status"><span class="spinner" aria-hidden="true"></span><span id="statusTxt">Abrindo painel…</span></div>
    </div>
  </div>
  <iframe id="painel" title="Painel Financeiro" allow="clipboard-read; clipboard-write"></iframe>

  <script>
    (() => {
      const LAUNCHER_VERSION = '13';
      const MOBILE_BUILD = '2.6';
      const APPS_SCRIPT = 'https://script.google.com/macros/s/AKfycbw0CRh3ZyFfQkjRz4vRhymlBd489xDWvswt4NlZt4nc_Lf6ZoDlTQh11OfTctiJaxnnTA/exec';
      const statusTxt = document.getElementById('statusTxt');

      function detectarCelular() {
        const ua = navigator.userAgent || '';
        const plataforma = navigator.platform || '';
        const pontosToque = navigator.maxTouchPoints || 0;

        const ehIPhone = /iPhone|iPod/i.test(ua);
        const ehIPad = /iPad/i.test(ua) || (plataforma === 'MacIntel' && pontosToque > 1);

        const ehAndroid = /Android/i.test(ua);
        const androidDizMobile = /Mobile/i.test(ua);

        const larguraViewport = Math.max(
          window.innerWidth || 0,
          document.documentElement.clientWidth || 0
        );

        const sw = Number(window.screen && window.screen.width) || 0;
        const sh = Number(window.screen && window.screen.height) || 0;
        const menorLadoTela = Math.min(sw || 9999, sh || 9999);
        const maiorLadoTela = Math.max(sw || 0, sh || 0);

        // iPad é sempre tratado como tablet.
        if (ehIPad) return false;

        // iPhone/iPod continuam na interface mobile.
        if (ehIPhone) return true;

        // Android sem "Mobile" no user-agent é tablet.
        if (ehAndroid && !androidDizMobile) return false;

        // Alguns tablets Android colocam "Mobile" no user-agent.
        // Nesse caso o tamanho real da tela tem prioridade sobre o user-agent.
        if (ehAndroid && androidDizMobile) {
          const pareceTablet =
            menorLadoTela >= 600 ||
            maiorLadoTela >= 900 ||
            larguraViewport >= 600;
          return !pareceTablet;
        }

        // Outros navegadores móveis conhecidos.
        if (/Windows Phone|IEMobile|Opera Mini|webOS|BlackBerry/i.test(ua)) return true;

        // Fallback: só classifica como telefone quando a viewport E a tela
        // realmente têm dimensões típicas de celular.
        return larguraViewport < 600 && menorLadoTela < 600;
      }

      function detectarSafariIOS() {
        const ua = navigator.userAgent || '';
        const iOS = /iPhone|iPad|iPod/i.test(ua) || (navigator.platform === 'MacIntel' && navigator.maxTouchPoints > 1);
        const safari = /Safari/i.test(ua) && !/CriOS|FxiOS|EdgiOS|OPiOS/i.test(ua);
        return iOS && safari;
      }

      async function limparCachesAntigos() {
        try {
          if ('serviceWorker' in navigator) {
            const regs = await navigator.serviceWorker.getRegistrations();
            await Promise.all(regs.map(r => r.unregister().catch(() => false)));
          }
          if ('caches' in window) {
            const keys = await caches.keys();
            await Promise.all(keys.filter(k => k.startsWith('tdb-launcher-')).map(k => caches.delete(k)));
          }
          try { localStorage.setItem('tdbLauncherVersion', LAUNCHER_VERSION); } catch (_) {}
        } catch (_) {}
      }

      const mobile = detectarCelular();
      const rota = mobile ? 'painelMobile' : 'painel';
      const frame = document.getElementById('painel');
      const splash = document.getElementById('splash');
      let liberado = false;
      let retryAutomatico = 0;

      function montarUrl(extra) {
        const qs = new URLSearchParams({
          acao: rota,
          pwa: '1',
          launcher: LAUNCHER_VERSION,
          build: mobile ? MOBILE_BUILD : 'desktop',
          _: Date.now().toString()
        });
        if (extra) Object.keys(extra).forEach(k => qs.set(k, String(extra[k])));
        return APPS_SCRIPT + '?' + qs.toString();
      }

      const revelar = () => {
        if (liberado) return;
        liberado = true;
        frame.classList.add('visivel');
        splash.classList.add('oculto');
      };

      async function iniciar() {
        if (statusTxt) statusTxt.textContent = mobile ? 'Abrindo versão celular…' : 'Abrindo painel…';
        await limparCachesAntigos();
        frame.src = montarUrl();
      }

      frame.addEventListener('load', revelar);
      setTimeout(revelar, 9000);

      // Se o painel reportar a falha temporária de abertura do Drive, o launcher
      // repete uma vez automaticamente. Assim o usuário não precisa tocar em
      // “Tentar novamente” quando a primeira resposta do Apps Script estiver fria/lenta.
      window.addEventListener('message', event => {
        const d = event && event.data;
        if (!d || d.type !== 'TDB_PAINEL_ERROR' || retryAutomatico >= 1) return;
        retryAutomatico += 1;
        if (statusTxt) statusTxt.textContent = 'Conexão lenta. Tentando novamente…';
        setTimeout(() => {
          frame.src = montarUrl({ retry: retryAutomatico });
        }, 700);
      });

      iniciar();
    })();
  </script>
